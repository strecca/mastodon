Using the stack at https://github.com/glitch-soc/mastodon I need to code a dynamic app with a dashboard UI that enables the admin to type in a name for a category of community resources, such as the Admin types 'artists' into an input with a "Create Category" button & that is step 1 to write a db/migrate/ 'artists'.rb postgres table with all of the fields & their possible types. But that migration file will not be written until the last step. After the Admin types in a short name for a new community category a schema.yml file is written to hold the values from a dynamically assembled Form editor. The Admin will see a Add Field input into which he/she types (example: 'first_name') & chooses type from a list of postgres types. The editor opens another input field for the Admin to type in another field name and choose the field type from a list. Then Save. On Save another input field opens below. When the Admin has finished with all fields the schema file is updated. THEN the UI opens the Form Editor where the Admin assigns further information such as whether a field will reflect option values in an array to be selected by the user via different types of checkboxes. When the full model of the Form is Saved then the schema will be updated to hold new information and the 'artists.yml' will be saved for later re-editing and the db/migrate/artist.rb file to migrate the new table to the database.

Architecture Overview
We’re adding a self-contained “Dynamic Community Resources” admin dashboard to your Glitch-Soc Mastodon instance (standard Rails + PostgreSQL stack).

    Schema storage: One YAML file per category at config/dynamic_schemas/{category}.yml (created on-the-fly, never touches db/schema.rb until you run a real migration).
    Two-phase editor (exactly as you described):
        “Basic fields” → name + raw PostgreSQL column type.
        “Form Editor” → UI metadata (label, widget, required, options array, etc.).
    Final step: On “Save Full Model”, the YAML is updated and a proper Rails migration file is auto-generated in db/migrate/. You (or the admin) then run rails db:migrate once.
    No dynamic models needed — after migration, you can use plain ActiveRecord (Artist.all, etc.) or a generic base class later.

class DynamicSchemaManager
  SCHEMA_DIR = Rails.root.join('config/dynamic_schemas')

  def self.path_for(category)
    SCHEMA_DIR.join("#{category.parameterize}.yml")
  end

  def self.load(category)
    path = path_for(category)
    return { 'category_name' => category, 'table_name' => category.pluralize, 'db_fields' => [], 'form_config' => { 'fields' => [] } } unless File.exist?(path)

    YAML.safe_load(File.read(path), permitted_classes: [Symbol]) || {}
  end

  def self.save(category, data)
    FileUtils.mkdir_p(SCHEMA_DIR)
    File.write(path_for(category), data.to_yaml)
  end

  def self.generate_migration!(schema)
    table = schema['table_name'] || schema['category_name'].pluralize
    timestamp = Time.current.strftime('%Y%m%d%H%M%S')
    migration_name = "create_#{table}"
    file_path = Rails.root.join("db/migrate/#{timestamp}_#{migration_name}.rb")

    fields_lines = schema['db_fields'].map do |f|
      col_name = f['name']
      col_type = f['type']

      if col_type.end_with?('[]')
        base = col_type[0..-3]
        "          t.#{base} :#{col_name}, array: true"
      else
        "          t.#{col_type} :#{col_name}"
      end
    end.join("\n")

    content = <<~RUBY
      # frozen_string_literal: true

      class #{migration_name.camelize} < ActiveRecord::Migration[7.1]  # ← Change to your exact Rails version from Gemfile if different
        def change
          create_table :#{table} do |t|
      #{fields_lines}
            t.timestamps
          end
        end
      end
    RUBY

    File.write(file_path, content)
    file_path.to_s
  end
end

app/controllers/admin/dynamic_categories_controller.rb

# frozen_string_literal: true

class Admin::DynamicCategoriesController < Admin::BaseController
  def index
    @categories = Dir[DynamicSchemaManager::SCHEMA_DIR.join('*.yml')].map do |f|
      File.basename(f, '.yml')
    end
  end

  def new; end

  def create
    category = params[:category_name].to_s.strip.parameterize
    return redirect_to admin_dynamic_categories_path, alert: 'Name required' if category.blank?

    schema = DynamicSchemaManager.load(category)
    DynamicSchemaManager.save(category, schema)
    redirect_to edit_admin_dynamic_category_path(category), notice: "Category '#{category}' created. Now add fields."
  end

  def edit
    @category = params[:id]
    @schema = DynamicSchemaManager.load(@category)
  end

  def save_fields
    category = params[:id]
    db_fields = params[:db_fields].permit!.to_h.values  # array of {name, type}
    schema = DynamicSchemaManager.load(category)
    schema['db_fields'] = db_fields
    DynamicSchemaManager.save(category, schema)

    render json: { success: true, next_step: 'form_editor' }
  end

  def save_form
    category = params[:id]
    schema = DynamicSchemaManager.load(category)

    # merge UI metadata into form_config
    schema['form_config'] ||= { 'fields' => [] }
    schema['form_config']['fields'] = params[:form_fields].permit!.to_h.values

    DynamicSchemaManager.save(category, schema)

    migration_path = DynamicSchemaManager.generate_migration!(schema)

    render json: {
      success: true,
      message: "Full model saved. Migration generated: #{migration_path}",
      instruction: 'Run `rails db:migrate` in your terminal to create the table.'
    }
  end
end

function saveFullForm() {
  // collect form fields from DOM
  const formData = Array.from(document.querySelectorAll('#form-fields-list > div')).map((div, i) => {
    const inputs = div.querySelectorAll('input, select, textarea');
    const obj = {};
    inputs.forEach(el => {
      if (el.type === 'checkbox') obj[el.name.split('[').pop().replace(']', '')] = el.checked;
      else if (el.name.includes('options')) obj.options = el.value.split('\n').map(l => l.trim()).filter(Boolean);
      else obj[el.name.split('[').pop().replace(']', '')] = el.value;
    });
    return obj;
  });

  fetch('<%= save_form_admin_dynamic_category_path(@category) %>', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json', 'X-CSRF-Token': document.querySelector('meta[name=csrf-token]').content },
    body: JSON.stringify({ form_fields: formData })
  })
  .then(r => r.json())
  .then(data => {
    if (data.success) {
      alert(data.message + '\n\n' + data.instruction);
      window.location.href = '<%= admin_dynamic_categories_path %>';
    }
  });
}

Here's how to integrate your dynamic community resources (e.g., Artists, Venues, Events) with Mastodon's Account model in Glitch-Soc.

This creates a clean, maintainable link so each Mastodon user (Account) can optionally have one or more community resource records. After migration, an admin or user can attach an "Artist profile" to their account, fill the dynamic form, and use it for directories, search, or public display.
Recommended Relationship: Polymorphic or Simple has_many

For flexibility across multiple categories (artists, musicians, nonprofits, etc.), we use a polymorphic association on a join-like table. But to keep it simple and fast:

    Each dynamic table (e.g., artists) gets an account_id column → belongs_to :account.
    The Account model gets has_many :artists (and similarly for other categories).

This is the most common and performant pattern in Mastodon forks.

class DynamicSchemaManager
  SCHEMA_DIR = Rails.root.join('config/dynamic_schemas')
  MODELS_DIR = Rails.root.join('app/models')

  def self.path_for(category)
    SCHEMA_DIR.join("#{category.parameterize}.yml")
  end

  def self.model_path_for(category)
    MODELS_DIR.join("#{category.parameterize}.rb")
  end

  def self.load(category)
    path = path_for(category)
    default = {
      'category_name' => category,
      'table_name' => category.pluralize.parameterize,
      'db_fields' => [],
      'form_config' => { 'fields' => [], 'link_to_account' => true }
    }
    return default unless File.exist?(path)

    YAML.safe_load(File.read(path), permitted_classes: [Symbol]) || default
  end

  def self.save(category, data)
    FileUtils.mkdir_p(SCHEMA_DIR)
    File.write(path_for(category), data.to_yaml)
  end

  def self.generate_migration!(schema)
    table = schema['table_name']
    timestamp = Time.current.strftime('%Y%m%d%H%M%S')
    migration_name = "create_#{table}"
    file_path = Rails.root.join("db/migrate/#{timestamp}_#{migration_name}.rb")

    fields_lines = schema['db_fields'].map do |f|
      col_name = f['name']
      col_type = f['type']

      if col_type.end_with?('[]')
        base = col_type[0..-3]
        "            t.#{base} :#{col_name}, array: true"
      else
        "            t.#{col_type} :#{col_name}"
      end
    end.join("\n")

    content = <<~RUBY
      # frozen_string_literal: true

      class #{migration_name.camelize} < ActiveRecord::Migration[7.1]
        def change
          create_table :#{table} do |t|
            t.references :account, null: false, foreign_key: true, index: true
      #{fields_lines}
            t.timestamps
          end
        end
      end
    RUBY

    File.write(file_path, content)
    file_path.to_s
  end

  # NEW: Auto-create the model file
  def self.generate_model!(schema)
    category = schema['category_name']
    model_name = category.camelize
    table_name = schema['table_name']
    model_path = model_path_for(category)

    # Build validations and attribute list from form_config + db_fields
    validations = []
    schema['form_config']['fields'].each do |field|
      next unless field['required']
      validations << "    validates :#{field['db_name']}, presence: true"
    end

    content = <<~RUBY
      # frozen_string_literal: true

      # Auto-generated dynamic model for community category: #{category}
      # Edit only if you need custom behavior. Re-saving the schema will overwrite this file.

      class #{model_name} < ApplicationRecord
        include DynamicResource

        self.table_name = "#{table_name}"

        # Validations from Form Editor
      #{validations.join("\n")}

        # Add custom scopes or methods here if needed
        # Example: scope :verified, -> { where(verified: true) }
      end
    RUBY

    FileUtils.mkdir_p(MODELS_DIR)
    File.write(model_path, content)
    model_path.to_s
  end
end

Step 2: Update the DynamicResource Concern (Add Search Scope)

Update app/models/concerns/dynamic_resource.rb:

# frozen_string_literal: true

module DynamicResource
  extend ActiveSupport::Concern

  included do
    belongs_to :account, class_name: 'Account', optional: false

    # Search scope: searches all text fields + account name/username
    scope :search, ->(query) {
      return all if query.blank?

      q = "%#{query.downcase}%"
      conditions = []

      # Search dynamic fields that are text/string
      text_columns = column_names.select { |c| [:string, :text].include?(columns_hash[c].type) }
      text_columns.each do |col|
        conditions << "#{col} ILIKE :q"
      end

      # Search array fields (e.g. genres[])
      array_columns = column_names.select { |c| columns_hash[c].array? }
      array_columns.each do |col|
        conditions << "EXISTS (SELECT 1 FROM unnest(#{col}) AS elem WHERE LOWER(elem) LIKE :q)"
      end

      # Search linked account
      joins(:account).where(
        conditions.any? ? conditions.join(' OR ') : '1=0',
        q: q
      ).or(
        joins(:account).where("accounts.display_name ILIKE :q OR accounts.username ILIKE :q", q: q)
      )
    }

    scope :publicly_visible, -> { where(public: true) } if column_names.include?('public')
  end
end

def autocomplete
    query = params[:q].to_s.strip
    return render json: [] if query.blank? || query.length < 2

    suggestions = []
    Dir[DynamicSchemaManager::SCHEMA_DIR.join('*.yml')].each do |file|
      cat = File.basename(file, '.yml')
      begin
        model = cat.camelize.constantize
        results = model
                    .publicly_visible
                    .search(query)
                    .includes(:account)
                    .limit(3)

        results.each do |res|
          display = "#{res.account.display_name || "@#{res.account.username}"} (#{cat.humanize})"
          suggestions << {
            label: display,
            value: query,  # keep original query for form submit
            category: cat,
            url: directory_category_path(cat, q: query)
          }
        end
      rescue
        next
      end
    end

    render json: suggestions.uniq { |s| s[:label] }.first(10)
  end

    else

    end
  end

    @category = params[:category]
    @resource = @category.camelize.constantize.where(account: current_account).first!
    @form_config = DynamicSchemaManager.load(@category)['form_config'] || { 'fields' => [] }
  end

    @category = params[:category]
    @resource = @category.camelize.constantize.where(account: current_account).first!
    if @resource.update(resource_params)

    else
      render :edit
    end

    permitted = DynamicSchemaManager.load(params[:category])['form_config']['fields'].map { |f| f['db_name'].to_sym }

  end

def index
  @categories = Dir[DynamicSchemaManager::SCHEMA_DIR.join('*.yml')]
                  .map { |f| File.basename(f, '.yml') }
                  .sort

  @query = params[:q].to_s.strip

  if @query.present?
    @global_results = {}   # directory resources
    @post_results = []     # Mastodon statuses from directory accounts

    # Directory resources (same as before)
    @categories.each do |cat|
      begin
        model = cat.camelize.constantize
        results = model.publicly_visible.search(@query).includes(:account).limit(3)
        @global_results[cat] = results if results.any?
      rescue
        next
      end
    end

    # Recent Mastodon posts from ANY account that has at least one directory entry
    if @global_results.any? || true  # or limit to accounts that appear in results
      directory_accounts = Account.joins(
        "INNER JOIN (#{all_dynamic_tables_union}) AS resources ON resources.account_id = accounts.id"
      ).distinct.limit(50)  # adjust as needed

      @post_results = Status
                        .where(account: directory_accounts)
                        .where(visibility: [:public, :unlisted])
                        .search(@query)  # Mastodon has built-in .search or use content ILIKE
                        .includes(:account)
                        .order(id: :desc)
                        .limit(10)
    end
  end
end

class DynamicResourceSerializer < ActiveModel::Serializer
  attributes :id, :created_at, :updated_at, :public

  belongs_to :account, serializer: AccountSerializer

  # Dynamically add all db_fields from the schema
  def attributes(*args)
    data = super
    form_config = object.class.name.underscore.pluralize
    schema = DynamicSchemaManager.load(form_config.chomp('s')) # rough mapping; improve if needed
    schema['db_fields']&.each do |field|
      data[field['name'].to_sym] = object.send(field['name'])
    end
    data
  end

  # Add form_config metadata for the frontend
  def form_config
    DynamicSchemaManager.load(object.class.name.underscore.pluralize.chomp('s'))['form_config']
  end
end

  case FETCH_CATEGORIES_REQUEST:
  case FETCH_RESOURCES_REQUEST:
    return state.set('loading', true).set('error', null);

      .set('loading', false);
  case FETCH_RESOURCES_SUCCESS:
    return state
      .set('resources', fromJS(action.resources))
      .set('formConfig', fromJS(action.formConfig))
      .set('loading', false);
  case FETCH_CATEGORIES_FAIL:
  case FETCH_RESOURCES_FAIL:
    return state.set('loading', false).set('error', action.error);
  default:
    return state;
  }
}

In modern React projects, the .erb file extension is not used as a standard part of the React build process. .erb

what do you mean by 'reducer'?

A "reducer" (in the context of Redux + React, which is what Glitch-Soc / Mastodon uses) is simply a plain JavaScript function that decides how the application's state should change.
Simple Explanation

Think of Redux as a central "brain" that holds all the data (state) for your feature — for example:

    The list of community directory categories
    The resources shown on a specific category page
    Whether it's currently loading
    Any search query or filters

When something happens in the app (user types in search, clicks a category, data loads from the server), the app doesn't change the data directly. Instead, it sends a message called an action (e.g., "FETCH_CATEGORIES_SUCCESS" with the new data attached).

The reducer is the function that listens to those actions and says:

    "Okay, here's the current state..."
    "Here's the action that just happened..."
    "Here's the new state I should return"

It always returns a new state object (it never mutates the old one directly — that's an important rule in Redux).
Real-World Analogy

Imagine a chef (the reducer):

    You give the chef the current ingredients on the counter (current state)
    You give the chef an instruction card (the action — e.g., "add tomatoes" or "bake for 20 minutes")
    The chef returns a new dish (new state) based on the old ingredients + the instruction

The chef doesn't change the original ingredients on the counter; he makes a fresh version.

// This is the reducer file
const initialState = {
  categories: [],
  resources: [],
  loading: false,
  error: null,
};

function communityDirectoryReducer(state = initialState, action) {
  switch (action.type) {
    case 'FETCH_CATEGORIES_SUCCESS':
      return { ...state, categories: action.categories, loading: false };
    case 'FETCH_RESOURCES_SUCCESS':
      return { ...state, resources: action.resources, loading: false };
    case 'FETCH_START':
      return { ...state, loading: true };
    default:
      return state;   // important: always return the current state if we don't care about this action
  }
}

  def self.load(category)

    return default_schema(category) unless File.exist?(path)

  end

// app/javascript/flavours/glitch/features/community_directory/components/dynamic_form_renderer.jsx
import React from 'react';

const DynamicFormRenderer = ({ formConfig, formData, onChange, isEdit = false }) => {
  const fields = formConfig.get('fields') || [];

  return (
    <>
      {fields.map((field, index) => {
        const name = field.get('db_name');
        const label = field.get('label') || name.humanize();
        const widget = field.get('widget') || 'text';
        const required = field.get('required');
        const options = field.get('options') || [];

        let inputElement;

        switch (widget) {
          case 'textarea':
            inputElement = (
              <textarea
                name={name}
                value={formData[name] || ''}
                onChange={(e) => onChange(name, e.target.value)}
                className="form-control"
                required={required}
              />
            );
            break;
          case 'select':
            inputElement = (
              <select
                name={name}
                value={formData[name] || ''}
                onChange={(e) => onChange(name, e.target.value)}
                className="form-control"
                required={required}
              >
                <option value="">Select...</option>
                {options.map((opt, i) => (
                  <option key={i} value={opt}>{opt}</option>
                ))}
              </select>
            );
            break;
          case 'checkboxes':
            inputElement = (
              <div>
                {options.map((opt, i) => (
                  <label key={i} className="form-check">
                    <input
                      type="checkbox"
                      checked={(formData[name] || []).includes(opt)}
                      onChange={(e) => {
                        const current = formData[name] || [];
                        const newVal = e.target.checked
                          ? [...current, opt]
                          : current.filter(v => v !== opt);
                        onChange(name, newVal);
                      }}
                    />
                    {opt}
                  </label>
                ))}
              </div>
            );
            break;
          default: // text, date, etc.
            inputElement = (
              <input
                type={widget === 'date' ? 'date' : 'text'}
                name={name}
                value={formData[name] || ''}
                onChange={(e) => onChange(name, e.target.value)}
                className="form-control"
                required={required}
              />
            );
        }

        return (
          <div key={index} className="field mb-3">
            <label className="form-label">
              {label} {required && <span className="text-danger">*</span>}
            </label>
            {inputElement}
          </div>
        );
      })}
    </>
  );
};

export default DynamicFormRenderer;

class Api::V1::CommunityDirectoryController < Api::BaseController
  before_action :set_category, only: [:category, :show, :create, :update]

  def index
    categories = Dir[DynamicSchemaManager::SCHEMA_DIR.join('*.yml')].map do |f|
      cat = File.basename(f, '.yml')
      schema = DynamicSchemaManager.load(cat)
      {
        name: cat,
        humanized: cat.humanize,
        icon: schema.dig('form_config', 'icon') || 'fa-users',
        table_name: schema['table_name']
      }
    end.sort_by { |c| c[:humanized] }

    render json: { categories: categories }
  end

  def category
    @query = params[:q].to_s.strip
    resources = @model.publicly_visible.search(@query).includes(:account).page(params[:page]).per(20)

    render json: {
      resources: ActiveModel::Serializer::CollectionSerializer.new(resources, serializer: DynamicResourceSerializer),
      form_config: @schema['form_config'],
      pagination: { current_page: resources.current_page, total_pages: resources.total_pages }
    }
  end

  def show
    resource = @model.find(params[:id])
    render json: DynamicResourceSerializer.new(resource)
  end

  def create
    resource = @model.new(resource_params)
    resource.account = current_user.account

    if resource.save
      render json: DynamicResourceSerializer.new(resource), status: :created
    else
      render json: { errors: resource.errors }, status: :unprocessable_entity
    end
  end

  def update
    resource = @model.where(account: current_user.account).find(params[:id])
    if resource.update(resource_params)
      render json: DynamicResourceSerializer.new(resource)
    else
      render json: { errors: resource.errors }, status: :unprocessable_entity
    end
  end

  private

  def set_category
    @category = params[:category]
    @schema = DynamicSchemaManager.load(@category)
    @model = @category.camelize.constantize
  rescue NameError
    render json: { error: 'Category not found' }, status: :not_found
  end

  def resource_params
    permitted = @schema.dig('form_config', 'fields')&.map { |f| f['db_name'].to_sym } || []
    params.require(@category.singularize.to_sym).permit(permitted + [:public])

// app/javascript/flavours/glitch/features/community_directory/components/dynamic_form_renderer.jsx
import React from 'react';

const DynamicFormRenderer = ({ formConfig, formData, onChange, isEdit = false }) => {
  const fields = formConfig.get('fields') || [];

  return (
    <>
      {fields.map((field, index) => {
        const name = field.get('db_name');
        const label = field.get('label') || name.humanize();
        const widget = field.get('widget') || 'text';
        const required = field.get('required') || false;
        const options = field.get('options') || [];

        let inputElement;

        switch (widget) {
          case 'textarea':
            inputElement = (
              <textarea
                name={name}
                value={formData[name] || ''}
                onChange={(e) => onChange(name, e.target.value)}
                className="form-control"
                required={required}
                rows={4}
              />
            );
            break;

          case 'select':
            inputElement = (
              <select
                name={name}
                value={formData[name] || ''}
                onChange={(e) => onChange(name, e.target.value)}
                className="form-control"
                required={required}
              >
                <option value="">Select an option...</option>
                {options.map((opt, i) => (
                  <option key={i} value={opt}>{opt}</option>
                ))}
              </select>
            );
            break;

          case 'checkboxes':
            inputElement = (
              <div className="checkbox-group">
                {options.map((opt, i) => (
                  <label key={i} className="form-check">
                    <input
                      type="checkbox"
                      checked={(formData[name] || []).includes(opt)}
                      onChange={(e) => {
                        const current = formData[name] || [];
                        const newVal = e.target.checked
                          ? [...current, opt]
                          : current.filter(v => v !== opt);
                        onChange(name, newVal);
                      }}
                    />
                    {opt}
                  </label>
                ))}
              </div>
            );
            break;

          default: // text, date, number, etc.
            inputElement = (
              <input
                type={widget === 'date' ? 'date' : 'text'}
                name={name}
                value={formData[name] || ''}
                onChange={(e) => onChange(name, e.target.value)}
                className="form-control"
                required={required}
              />
            );
        }

        return (
          <div key={index} className="field mb-4">
            <label className="form-label">
              {label}
              {required && <span className="text-danger"> *</span>}
            </label>
            {inputElement}
          </div>
        );
      })}
    </>
  );
};

        .set('loading', false);

    case 'COMMUNITY_DIRECTORY_UPDATE_RESOURCE_SUCCESS':
      return state.set('loading', false);

    case 'COMMUNITY_DIRECTORY_FETCH_RESOURCES_FAIL':
    case 'COMMUNITY_DIRECTORY_FETCH_RESOURCE_FAIL':
    case 'COMMUNITY_DIRECTORY_CREATE_RESOURCE_FAIL':

class Api::V1::CommunityDirectoryController < Api::BaseController
  before_action :set_category, only: [:category, :show, :create, :update]
  before_action :authenticate_user!, only: [:create, :update]

  # GET /api/v1/community_directory
  def index
    categories = Dir[DynamicSchemaManager::SCHEMA_DIR.join('*.yml')].map do |f|
      cat = File.basename(f, '.yml')
      schema = DynamicSchemaManager.load(cat)
      {
        name: cat,
        humanized: cat.humanize,
        icon: schema.dig('form_config', 'icon') || 'fa-users',
        table_name: schema['table_name']
      }
    end.sort_by { |c| c[:humanized] }

    render json: { categories: categories }
  end

  # GET /api/v1/community_directory/:category
  def category
    @query = params[:q].to_s.strip

    resources = @model
                  .publicly_visible
                  .search(@query)
                  .includes(:account)
                  .order(created_at: :desc)
                  .page(params[:page] || 1)
                  .per(24)

    render json: {
      resources: ActiveModel::Serializer::CollectionSerializer.new(
        resources,
        serializer: DynamicResourceSerializer
      ),
      form_config: @schema['form_config'] || {},
      pagination: {
        current_page: resources.current_page,
        total_pages: resources.total_pages,
        total_count: resources.total_count
      }
    }
  end

  # GET /api/v1/community_directory/:category/:id
  def show
    resource = @model.find(params[:id])
    render json: DynamicResourceSerializer.new(resource)
  end

# app/models/concerns/dynamic_resource.rb

module DynamicResource
  extend ActiveSupport::Concern

  included do
    belongs_to :account, class_name: 'Account', optional: false

    # Default public visibility (add 'public: boolean' field in admin editor if you want control)
    scope :publicly_visible, -> { where(public: true) } if column_names.include?('public')

    # Optional: Add a simple search scope (used by the API)
    scope :search, ->(query) {
      return all if query.blank?

      q = "%#{query.downcase}%"
      conditions = []

      # Search text/string columns
      text_columns = column_names.select { |c| [:string, :text].include?(columns_hash[c]&.type) }
      text_columns.each do |col|
        conditions << "#{col} ILIKE :q"
      end

      # Search array columns (e.g. genres[])
      array_columns = column_names.select { |c| columns_hash[c]&.array? }
      array_columns.each do |col|
        conditions << "EXISTS (SELECT 1 FROM unnest(#{col}) AS elem WHERE LOWER(elem) LIKE :q)"
      end

      # Search linked account name/username
      joins(:account).where(
        conditions.any? ? conditions.join(' OR ') : '1=0',
        q: q
      ).or(
        joins(:account).where("accounts.display_name ILIKE :q OR accounts.username ILIKE :q", q: q)
      )
    }
  end
end

// app/javascript/flavours/glitch/features/community_directory/components/dynamic_form_renderer.jsx
import React from 'react';

const DynamicFormRenderer = ({ formConfig, formData, onChange }) => {
  const fields = formConfig.get('fields') || [];

  return (
    <>
      {fields.map((field, index) => {
        const name = field.get('db_name');
        const label = field.get('label') || name.humanize();
        const widget = field.get('widget') || 'text';
        const required = field.get('required') || false;
        const options = field.get('options') || [];

        let inputElement;

        switch (widget) {
          case 'textarea':
            inputElement = (
              <textarea
                name={name}
                value={formData[name] || ''}
                onChange={(e) => onChange(name, e.target.value)}
                className="form-control"
                required={required}
                rows={4}
              />
            );
            break;

          case 'select':
            inputElement = (
              <select
                name={name}
                value={formData[name] || ''}
                onChange={(e) => onChange(name, e.target.value)}
                className="form-control"
                required={required}
              >
                <option value="">Select an option...</option>
                {options.map((opt, i) => (
                  <option key={i} value={opt}>{opt}</option>
                ))}
              </select>
            );
            break;

          case 'checkboxes':
            inputElement = (
              <div className="checkbox-group">
                {options.map((opt, i) => (
                  <label key={i} className="form-check">
                    <input
                      type="checkbox"
                      checked={(formData[name] || []).includes(opt)}
                      onChange={(e) => {
                        const current = formData[name] || [];
                        const newVal = e.target.checked
                          ? [...current, opt]
                          : current.filter(v => v !== opt);
                        onChange(name, newVal);
                      }}
                    />
                    {opt}
                  </label>
                ))}
              </div>
            );
            break;

          default: // text, date, etc.
            inputElement = (
              <input
                type={widget === 'date' ? 'date' : 'text'}
                name={name}
                value={formData[name] || ''}
                onChange={(e) => onChange(name, e.target.value)}
                className="form-control"
                required={required}
              />
            );
        }

        return (
          <div key={index} className="field mb-4">
            <label className="form-label">
              {label}
              {required && <span className="text-danger"> *</span>}
            </label>
            {inputElement}
          </div>
        );
      })}
    </>
  );
};

    'unknown'
  end

  def safe_metadata
    schema = DynamicSchemaManager.load(category_name)
    fields = schema.dig('form_config', 'fields') || []

    fields.map do |field|
      {
        db_name: field['db_name'] || '',
        label: field['label'] || (field['db_name'] || '').humanize,
        widget: field['widget'] || 'text',
        required: field['required'] || false,
        searchable: field['searchable'] || false,
        options: field['options'] || [],
        type: safe_column_type(field['db_name'])
      }
    end
  rescue StandardError => e
    Rails.logger.error "Metadata generation error: #{e.message}"
    []
  end

  def safe_form_config
    DynamicSchemaManager.load(category_name)['form_config'] || {}
  rescue StandardError
    {}
  end

  def safe_column_type(db_name)
    column = object.class.columns_hash[db_name]
    return nil unless column
    column.array? ? "#{column.type}[]" : column.type.to_s
  rescue StandardError
    'string'
  end

  def display_name
    object.try(:title) ||
      object.try(:name) ||
      [object.try(:first_name), object.try(:last_name)].compact.join(' ').presence ||
      "#{object.account&.display_name || 'User'}'s #{category_name.humanize}"
  rescue StandardError
    'Entry'
  end

  def public
    object.respond_to?(:public) ? !!object.public : true
  rescue StandardError
    true
  end
end

// app/javascript/flavours/glitch/features/community_directory/components/dynamic_form_renderer.jsx
import React from 'react';

const DynamicFormRenderer = ({ formConfig, formData, onChange, errors = {} }) => {
  const fields = formConfig.get('fields') || [];

  return (
    <>
      {fields.map((field, index) => {
        const name = field.get('db_name');
        const label = field.get('label') || name.humanize();
        const widget = field.get('widget') || 'text';
        const required = field.get('required') || false;
        const options = field.get('options') || [];
        const fieldError = errors[name];

        let inputElement;

        switch (widget) {
          case 'textarea':
            inputElement = (
              <textarea
                name={name}
                value={formData[name] || ''}
                onChange={(e) => onChange(name, e.target.value)}
                className={`form-control ${fieldError ? 'is-invalid' : ''}`}
                required={required}
                rows={4}
              />
            );
            break;

          case 'select':
            inputElement = (
              <select
                name={name}
                value={formData[name] || ''}
                onChange={(e) => onChange(name, e.target.value)}
                className={`form-control ${fieldError ? 'is-invalid' : ''}`}
                required={required}
              >
                <option value="">Select an option...</option>
                {options.map((opt, i) => (
                  <option key={i} value={opt}>{opt}</option>
                ))}
              </select>
            );
            break;

          case 'checkboxes':
            inputElement = (
              <div className={`checkbox-group ${fieldError ? 'is-invalid' : ''}`}>
                {options.map((opt, i) => (
                  <label key={i} className="form-check">
                    <input
                      type="checkbox"
                      checked={(formData[name] || []).includes(opt)}
                      onChange={(e) => {
                        const current = formData[name] || [];
                        const newVal = e.target.checked
                          ? [...current, opt]
                          : current.filter(v => v !== opt);
                        onChange(name, newVal);
                      }}
                    />
                    {opt}
                  </label>
                ))}
              </div>
            );
            break;

          default: // text, date, etc.
            inputElement = (
              <input
                type={widget === 'date' ? 'date' : 'text'}
                name={name}
                value={formData[name] || ''}
                onChange={(e) => onChange(name, e.target.value)}
                className={`form-control ${fieldError ? 'is-invalid' : ''}`}
                required={required}
              />
            );
        }

        return (
          <div key={index} className="field mb-4">
            <label className="form-label">
              {label}
              {required && <span className="text-danger"> *</span>}
            </label>
            {inputElement}
            {fieldError && <div className="invalid-feedback">{fieldError}</div>}
          </div>
        );
      })}
    </>
  );
};

import { useDispatch, useSelector } from 'react-redux';

const CommunityDirectoryNew = ({ match }) => {

  const dispatch = useDispatch();
  const formConfig = useSelector(state => state.getIn(['community_directory', 'formConfig'], {}));
  const loading = useSelector(state => state.getIn(['community_directory', 'loading'], false));
  const error = useSelector(state => state.getIn(['community_directory', 'error']));
  const [formData, setFormData] = useState({});
  const [formErrors, setFormErrors] = useState({});

  useEffect(() => {
    if (category) {
      dispatch(fetchResources(category));
    }
  }, [dispatch, category]);

  const handleChange = (name, value) => {
    setFormData(prev => ({ ...prev, [name]: value }));
    // Clear error when user starts typing
    if (formErrors[name]) {
      setFormErrors(prev => ({ ...prev, [name]: null }));
    }
  };

  const validateForm = () => {
    const errors = {};
    const fields = formConfig.get('fields') || [];

    fields.forEach(field => {
      const name = field.get('db_name');
      const required = field.get('required');

          errors[name] = 'This field is required';
        }
      }
    });

    setFormErrors(errors);
    return Object.keys(errors).length === 0;
  };

  const handleSubmit = (e) => {
    e.preventDefault();
    if (!validateForm()) return;

    if (category) {
      dispatch(createResource(category, formData));
    }
  };

  if (error) return <div className="status error">Error: {error}</div>;
  if (!category) return <div className="status error">Category not specified.</div>;

          console.warn(`Error populating field ${field.get('db_name')}:`, e);
        }
      });
      setFormData(data);
    }
  }, [resource]);

  const handleChange = (name, value) => {
    setFormData(prev => ({ ...prev, [name]: value }));
    // Clear specific field error on change
    if (formErrors[name]) {
      setFormErrors(prev => ({ ...prev, [name]: null }));
    }
  };

  const validateForm = () => {
    const errors = {};
    const fields = formConfig.get('fields') || [];

    fields.forEach(field => {
      const name = field.get('db_name');
      const required = field.get('required') || false;
      const value = formData[name];

      if (required) {
        if (Array.isArray(value)) {
          if (value.length === 0) {
            errors[name] = 'This field is required';
          }
        } else if (!value || String(value).trim() === '') {
          errors[name] = 'This field is required';
        }
      }
    });

    setFormErrors(errors);
    return Object.keys(errors).length === 0;
  };

  const handleSubmit = (e) => {
    e.preventDefault();
    if (!validateForm()) return;

    if (category && id) {
      dispatch(updateResource(category, id, formData));
    }
  };

  // Error / loading states
  if (error) {
    return <div className="status error">Error loading edit form: {error}</div>;
  }

  if (loading || !resource) {
    return <div className="loading-bar">Loading edit form...</div>;
  }

  if (!category) {
    return <div className="status error">Category not specified.</div>;
  }

  return (
    <div>
      <h1>Edit Your {category.humanize().singularize} Entry</h1>
      <form onSubmit={handleSubmit}>
        <DynamicFormRenderer 
          formConfig={formConfig} 
          formData={formData} 
          onChange={handleChange}
          errors={formErrors}
        />
        <button 
          type="submit" 
          className="btn btn-primary" 
          disabled={loading}
        >
          {loading ? 'Saving Changes...' : 'Save Changes'}
        </button>
      </form>
    </div>
  );
};

class Api::V1::CommunityDirectoryController < Api::BaseController
  before_action :set_category, only: [:category, :show, :create, :update]
  before_action :authenticate_user!, only: [:create, :update]

  # GET /api/v1/community_directory
  def index
    categories = Dir[DynamicSchemaManager::SCHEMA_DIR.join('*.yml')].map do |f|
      cat = File.basename(f, '.yml')
      schema = DynamicSchemaManager.load(cat)
      {
        name: cat,
        humanized: cat.humanize,
        icon: schema.dig('form_config', 'icon') || 'fa-users',
        table_name: schema['table_name']
      }
    end.sort_by { |c| c[:humanized] }

    render json: { categories: categories }
  end

  # GET /api/v1/community_directory/:category
  def category
    @query = params[:q].to_s.strip

    resources = @model.publicly_visible.includes(:account)

    # 1. Text search (using existing search scope)
    resources = resources.search(@query) if @query.present?

    # 2. Server-side filters from ?filters[field_name]=true
    if params[:filters].present?
      searchable_fields = @schema.dig('form_config', 'fields')&.select { |f| f['searchable'] } || []

      params[:filters].permit!.each do |field_name, value|
        # Only allow filters on fields marked searchable in admin schema
        if searchable_fields.any? { |f| f['db_name'] == field_name } && value.present?
          resources = resources.where(field_name => ActiveModel::Type::Boolean.new.cast(value))
        end
      end
    end

    resources = resources.order(created_at: :desc)
                         .page(params[:page] || 1)
                         .per(24)

    render json: {
      resources: ActiveModel::Serializer::CollectionSerializer.new(
        resources,
        serializer: DynamicResourceSerializer
      ),
      form_config: @schema['form_config'] || {},
      pagination: {
        current_page: resources.current_page,
        total_pages: resources.total_pages,
        total_count: resources.total_count
      }
    }
  rescue StandardError => e
    Rails.logger.error "CommunityDirectoryController#category error: #{e.message}"
    render json: { error: 'Failed to load resources' }, status: :internal_server_error
  end

  # GET /api/v1/community_directory/:category/:id
  def show
    resource = @model.find(params[:id])
    render json: DynamicResourceSerializer.new(resource)
  rescue ActiveRecord::RecordNotFound
    render json: { error: 'Resource not found' }, status: :not_found
  end

  # POST /api/v1/community_directory/:category
  def create
    resource = @model.new(resource_params)
    resource.account = current_user.account

    if resource.save
      render json: DynamicResourceSerializer.new(resource), status: :created
    else
      render json: { errors: resource.errors.full_messages }, status: :unprocessable_entity
    end
  end

  # PATCH /api/v1/community_directory/:category/:id
  def update
    resource = @model.where(account: current_user.account).find(params[:id])

    if resource.update(resource_params)
      render json: DynamicResourceSerializer.new(resource)
    else
      render json: { errors: resource.errors.full_messages }, status: :unprocessable_entity
    end
  end

  private

  def set_category
    @category = params[:category]
    @schema = DynamicSchemaManager.load(@category)
    @model = @category.camelize.constantize
  rescue NameError
    render json: { error: "Category '#{@category}' not found" }, status: :not_found
  end

  def resource_params
    permitted_fields = @schema.dig('form_config', 'fields')&.map { |f| f['db_name'].to_sym } || []
    params.require(@category.singularize.to_sym).permit(permitted_fields + [:public])
  end

// app/javascript/flavours/glitch/features/community_directory/components/dynamic_form_renderer.jsx
import React from 'react';

const DynamicFormRenderer = ({ formConfig, formData, onChange, errors = {} }) => {
  const fields = formConfig.get('fields') || [];

  return (
    <>
      {fields.map((field, index) => {
        const name = field.get('db_name');
        const label = field.get('label') || name.humanize();
        const widget = field.get('widget') || 'text';
        const required = field.get('required') || false;
        const options = field.get('options') || [];
        const fieldError = errors[name];

        let inputElement;

        switch (widget) {
          case 'textarea':
            inputElement = (
              <textarea
                name={name}
                value={formData[name] || ''}
                onChange={(e) => onChange(name, e.target.value)}
                className={`form-control ${fieldError ? 'is-invalid' : ''}`}
                required={required}
                rows={4}
              />
            );
            break;

          case 'select':
            inputElement = (
              <select
                name={name}
                value={formData[name] || ''}
                onChange={(e) => onChange(name, e.target.value)}
                className={`form-control ${fieldError ? 'is-invalid' : ''}`}
                required={required}
              >
                <option value="">Select an option...</option>
                {options.map((opt, i) => (
                  <option key={i} value={opt}>{opt}</option>
                ))}
              </select>
            );
            break;

          case 'checkboxes':
            inputElement = (
              <div className={`checkbox-group ${fieldError ? 'is-invalid' : ''}`}>
                {options.map((opt, i) => (
                  <label key={i} className="form-check">
                    <input
                      type="checkbox"
                      checked={(formData[name] || []).includes(opt)}
                      onChange={(e) => {
                        const current = formData[name] || [];
                        const newVal = e.target.checked
                          ? [...current, opt]
                          : current.filter(v => v !== opt);
                        onChange(name, newVal);
                      }}
                    />
                    {opt}
                  </label>
                ))}
              </div>
            );
            break;

          default: // text, date, number, etc.
            inputElement = (
              <input
                type={widget === 'date' ? 'date' : 'text'}
                name={name}
                value={formData[name] || ''}
                onChange={(e) => onChange(name, e.target.value)}
                className={`form-control ${fieldError ? 'is-invalid' : ''}`}
                required={required}
              />
            );
        }

        return (
          <div key={index} className="field mb-4">
            <label className="form-label">
              {label}
              {required && <span className="text-danger"> *</span>}
            </label>
            
            {inputElement}
            
            {/* Validation error message */}
            {fieldError && (
              <div className="invalid-feedback d-block">
                {fieldError}
              </div>
            )}
          </div>
        );
      })}
    </>
  );
};

// app/javascript/flavours/glitch/features/community_directory/components/dynamic_form_renderer.jsx
import React from 'react';

const DynamicFormRenderer = ({ formConfig, formData, onChange, errors = {} }) => {
  const fields = formConfig.get('fields') || [];

  return (
    <>
      {fields.map((field, index) => {
        const name = field.get('db_name');
        const label = field.get('label') || name.humanize();
        const widget = field.get('widget') || 'text';
        const required = field.get('required') || false;
        const options = field.get('options') || [];
        const fieldError = errors[name];

        let inputElement;

        switch (widget) {
          case 'textarea':
            inputElement = (
              <textarea
                name={name}
                value={formData[name] || ''}
                onChange={(e) => onChange(name, e.target.value)}
                className={`form-control ${fieldError ? 'is-invalid' : ''}`}
                required={required}
                rows={4}
              />
            );
            break;

          case 'select':
            inputElement = (
              <select
                name={name}
                value={formData[name] || ''}
                onChange={(e) => onChange(name, e.target.value)}
                className={`form-control ${fieldError ? 'is-invalid' : ''}`}
                required={required}
              >
                <option value="">Select an option...</option>
                {options.map((opt, i) => (
                  <option key={i} value={opt}>{opt}</option>
                ))}
              </select>
            );
            break;

          case 'checkboxes':
            inputElement = (
              <div className={`checkbox-group ${fieldError ? 'is-invalid' : ''}`}>
                <div className="checkbox-group-label">
                  {label} {required && <span className="text-danger">*</span>}
                </div>
                {options.map((opt, i) => (
                  <label key={i} className="form-check form-check-inline">
                    <input
                      type="checkbox"
                      checked={(formData[name] || []).includes(opt)}
                      onChange={(e) => {
                        const current = formData[name] || [];
                        const newVal = e.target.checked
                          ? [...current, opt]
                          : current.filter(v => v !== opt);
                        onChange(name, newVal);
                      }}
                    />
                    <span className="form-check-label">{opt}</span>
                  </label>
                ))}
                {fieldError && <div className="invalid-feedback d-block mt-1">{fieldError}</div>}
              </div>
            );
            break;

          case 'radio':
            // New: Radio button support
            inputElement = (
              <div className={`radio-group ${fieldError ? 'is-invalid' : ''}`}>
                <div className="radio-group-label">
                  {label} {required && <span className="text-danger">*</span>}
                </div>
                {options.map((opt, i) => (
                  <label key={i} className="form-check form-check-inline">
                    <input
                      type="radio"
                      name={name}
                      value={opt}
                      checked={formData[name] === opt}
                      onChange={(e) => onChange(name, e.target.value)}
                      className={fieldError ? 'is-invalid' : ''}
                    />
                    <span className="form-check-label">{opt}</span>
                  </label>
                ))}
                {fieldError && <div className="invalid-feedback d-block mt-1">{fieldError}</div>}
              </div>
            );
            break;

          default: // text, date, etc.
            inputElement = (
              <input
                type={widget === 'date' ? 'date' : 'text'}
                name={name}
                value={formData[name] || ''}
                onChange={(e) => onChange(name, e.target.value)}
                className={`form-control ${fieldError ? 'is-invalid' : ''}`}
                required={required}
              />
            );
        }

        // Render non-checkbox/radio fields
        if (widget !== 'checkboxes' && widget !== 'radio') {
          return (
            <div key={index} className="field mb-4">
              <label className="form-label">
                {label}
                {required && <span className="text-danger"> *</span>}
              </label>
              {inputElement}
              {fieldError && <div className="invalid-feedback d-block">{fieldError}</div>}
            </div>
          );
        }

        // Checkbox and Radio groups are self-contained
        return <div key={index} className="mb-4">{inputElement}</div>;
      })}
    </>
  );
};

export default DynamicFormRenderer;
